7: /-- The genuine source weaker-norm estimate turns L2 convergence of data into
8: coefficient convergence of their continuous source flows. -/
9: theorem source_flow_coefficient_limit (hL : BITWeakerLipschitz)
10:     {T M : ℝ} (hT : 0 < T) (hM : 0 ≤ M)
11:     {data : ℕ → Fourier} {f : Fourier} {flows : ℕ → ℝ → Fourier} {v : ℝ → Fourier}
12:     (hdataR : ∀ n, RealValued (data n)) (hfR : RealValued f)
13:     (hdata : ∀ n, InHomSobolev 0 (data n)) (hf : InHomSobolev 0 f)
14:     (hbound : ∀ n, homogeneousNorm 0 (data n) + homogeneousNorm 0 f ≤ M)
15:     (hflows : ∀ n, SourceWeakSolution T (data n) (flows n))
16:     (hflowsC : ∀ n, HomSobolevContinuousOn 0 T (flows n))
17:     (hv : SourceWeakSolution T f v) (hvC : HomSobolevContinuousOn 0 T v)
18:     (hd : Filter.Tendsto (fun n => homogeneousNorm 0 (data n - f)) Filter.atTop (nhds 0)) :
19:     ∀ t ∈ Set.Icc 0 T, ∀ k : ℤ,
20:       Filter.Tendsto (fun n => flows n t k) Filter.atTop (nhds (v t k)) := by
21:   obtain ⟨C, hC, A, hA, hb⟩ := source_flow_lipschitz hL hM
22:     (by norm_num : (-1 : ℝ) < 0) (le_refl (0 : ℝ))
23:   intro t ht k
24:   have hcoeff (n : ℕ) : ‖flows n t k - v t k‖ ≤
25:       (C * A ^ T) * homogeneousNorm 0 (data n - f) := by
26:     have hdiff : InSobolev 0 (flows n t - v t) :=
27:       (((hflowsC n).1 t ht).sub (hvC.1 t ht)).to_inhomogeneous (le_refl 0)
28:     have hc : ‖flows n t k - v t k‖ ≤ homogeneousNorm 0 (flows n t - v t) := by
29:       simpa only [homogeneousNorm_zero, weighted_zero, Pi.sub_apply] using weighted_coeff_bound hdiff k
30:     exact hc.trans (hb T hT (data n) f (hdataR n) hfR (hdata n) hf (hbound n)
31:       (flows n) v (hflows n) hv (hflowsC n) hvC t ht)
32:   rw [tendsto_iff_norm_sub_tendsto_zero]
33:   apply squeeze_zero (fun n => norm_nonneg _) hcoeff
34:   simpa using hd.const_mul (C * A ^ T)

37: /-- Project-specific consequence: estimates with constants only on exact norm
38: spheres imply the historical neighborhood estimate. All flow and estimate
39: hypotheses are explicit here; none is asserted as a literature theorem. -/

45:       ∀ f g : Fourier, RealValued f → RealValued g → InHomSobolev s f → InHomSobolev s g →
46:       homogeneousNorm s f = S → homogeneousNorm s g = S →
47:       ∀ v w : ℝ → Fourier, SourceWeakSolution T f v → SourceWeakSolution T g w →
