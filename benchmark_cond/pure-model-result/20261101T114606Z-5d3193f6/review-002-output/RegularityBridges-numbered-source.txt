178: /-- The literal iteration factor in (6.7) has the usual exponential-in-time
179: form with constants independent of T whenever its governing bound is fixed. -/
180: theorem iteration_factor_le_exponential {T τ C : ℝ} (hT : 0 ≤ T) (hτ : 0 < τ) (hC : 0 < C) :
181:     (2 * C + 1) ^ (⌊T / τ⌋₊ + 1) ≤
182:       (2 * C + 1) * ((2 * C + 1) ^ (1 / τ : ℝ)) ^ T := by
183:   have hB : 0 < 2 * C + 1 := by linarith
184:   have hB1 : 1 ≤ 2 * C + 1 := by linarith
185:   calc
186:     (2 * C + 1) ^ (⌊T / τ⌋₊ + 1) =
187:         (2 * C + 1) ^ ((⌊T / τ⌋₊ : ℝ) + 1 : ℝ) := by
188:       rw [← Real.rpow_natCast]
189:       simp only [Nat.cast_add, Nat.cast_one]
190:     _ ≤ (2 * C + 1) ^ (T / τ + 1 : ℝ) :=
191:       Real.rpow_le_rpow_of_exponent_le hB1 (by linarith [Nat.floor_le (div_nonneg hT hτ.le)])
192:     _ = (2 * C + 1) * ((2 * C + 1) ^ (1 / τ : ℝ)) ^ T := by
193:       rw [Real.rpow_add hB, Real.rpow_one, ← Real.rpow_mul hB.le]
194:       rw [mul_comm ((2 * C + 1) ^ (T / τ : ℝ)) (2 * C + 1)]
195:       congr 1
196:       congr 1
197:       ring
