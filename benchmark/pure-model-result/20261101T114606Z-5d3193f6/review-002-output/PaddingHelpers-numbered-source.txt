7: /-- The amplitude used to lift a radius-R ball to the radius-2R sphere is
8: Lipschitz. This scalar estimate controls the *difference* of the two paddings. -/
9: theorem sqrt_padding_lipschitz {R x y d : ℝ} (hR : 0 < R)
10:     (hx : 0 ≤ x) (hxR : x ≤ R) (hy : 0 ≤ y) (hyR : y ≤ R)
11:     (hd : 0 ≤ d) (hxy : |x - y| ≤ d) :
12:     |Real.sqrt (((2 * R) ^ 2 - x ^ 2) / 2) -
13:       Real.sqrt (((2 * R) ^ 2 - y ^ 2) / 2)| ≤ d := by
14:   let a := Real.sqrt (((2 * R) ^ 2 - x ^ 2) / 2)
15:   let b := Real.sqrt (((2 * R) ^ 2 - y ^ 2) / 2)
16:   have hradx : 0 ≤ ((2 * R) ^ 2 - x ^ 2) / 2 := by nlinarith
17:   have hrady : 0 ≤ ((2 * R) ^ 2 - y ^ 2) / 2 := by nlinarith
18:   have ha : a ^ 2 = ((2 * R) ^ 2 - x ^ 2) / 2 := Real.sq_sqrt hradx
19:   have hb : b ^ 2 = ((2 * R) ^ 2 - y ^ 2) / 2 := Real.sq_sqrt hrady
20:   have ha0 : 0 ≤ a := Real.sqrt_nonneg _
21:   have hb0 : 0 ≤ b := Real.sqrt_nonneg _
22:   have haR : R ≤ a := by nlinarith
23:   have hbR : R ≤ b := by nlinarith
24:   have hsum : x + y ≤ 2 * R := by linarith
25:   obtain ⟨hxy1, hxy2⟩ := abs_le.mp hxy
26:   have upper : ∀ a b x y : ℝ,
27:       a ^ 2 = ((2 * R) ^ 2 - x ^ 2) / 2 →
28:       b ^ 2 = ((2 * R) ^ 2 - y ^ 2) / 2 →
29:       R ≤ a → R ≤ b → 0 ≤ x + y → x + y ≤ 2 * R → y - x ≤ d → a - b ≤ d := by
30:     intro a b x y ha hb haR hbR hsum0 hsum hyxd
31:     by_cases hab : a ≤ b
32:     · linarith
33:     have hab0 : 0 ≤ a - b := by linarith
34:     have hm1 := mul_nonneg (sub_nonneg.mpr hsum) hd
35:     have hm2 := mul_nonneg hsum0 (sub_nonneg.mpr hyxd)
36:     have hm3 := mul_nonneg hab0 (show 0 ≤ a + b - 2 * R by linarith)
37:     have hRd : 0 ≤ R * d := mul_nonneg hR.le hd
38:     apply (mul_le_mul_iff_right₀ hR).mp
39:     nlinarith
40:   apply abs_le.mpr
41:   constructor
42:   · have h := upper b a y x hb ha hbR haR (by linarith) (by linarith) hxy2
43:     linarith
44:   · exact upper a b x y ha hb haR hbR (by linarith) hsum (by linarith)

49: /-- The weighted sequence before homogeneous reconstruction. -/
50: def pairedSphereLift (p : ℤ) (R : ℝ) (F : L2Seq) : L2Seq :=
51:   F + symmetricSpike p (Real.sqrt (((2 * R) ^ 2 - ‖F‖ ^ 2) / 2))

53: theorem symmetricSpike_sub (p : ℤ) (a b : ℝ) :
54:     symmetricSpike p a - symmetricSpike p b = symmetricSpike p (a - b) := by
55:   apply Subtype.ext
56:   funext k
57:   simp only [symmetricSpike, lp.coeFn_sub, lp.coeFn_add, Pi.sub_apply, Pi.add_apply,
58:     lp.single_apply, Pi.single_apply]
59:   split_ifs <;> push_cast <;> ring

61: theorem symmetricSpike_norm_le (p : ℤ) (a : ℝ) :
62:     ‖symmetricSpike p a‖ ≤ 2 * |a| := by
63:   calc
64:     _ ≤ ‖(lp.single 2 p (a : ℂ) : L2Seq)‖ + ‖(lp.single 2 (-p) (a : ℂ) : L2Seq)‖ := norm_add_le _ _
65:     _ = _ := by rw [lp.norm_single (by norm_num : (0 : ℝ≥0∞) < 2),
66:       lp.norm_single (by norm_num : (0 : ℝ≥0∞) < 2)]; simp [Complex.norm_real, Real.norm_eq_abs]; ring

68: /-- A single fixed-radius sphere estimate suffices for comparison on the ball:
69: the simultaneous lift has a uniform bound for differences. -/
70: theorem pairedSphereLift_lipschitz {R : ℝ} (hR : 0 < R) (p : ℤ)
71:     (F G : L2Seq) (hF : ‖F‖ ≤ R) (hG : ‖G‖ ≤ R) :
72:     ‖pairedSphereLift p R F - pairedSphereLift p R G‖ ≤ 3 * ‖F - G‖ := by
73:   let a := Real.sqrt (((2 * R) ^ 2 - ‖F‖ ^ 2) / 2)
74:   let b := Real.sqrt (((2 * R) ^ 2 - ‖G‖ ^ 2) / 2)
75:   have hab : |a - b| ≤ ‖F - G‖ := sqrt_padding_lipschitz hR
76:     (norm_nonneg F) hF (norm_nonneg G) hG (norm_nonneg _) (abs_norm_sub_norm_le F G)
77:   have he : pairedSphereLift p R F - pairedSphereLift p R G =
78:       (F - G) + symmetricSpike p (a - b) := by
79:     change (F + symmetricSpike p a) - (G + symmetricSpike p b) = _
80:     rw [← symmetricSpike_sub p a b]
81:     exact add_sub_add_comm _ _ _ _
82:   have htri := norm_add_le (F - G) (symmetricSpike p (a - b))
83:   have hsp := symmetricSpike_norm_le p (a - b)
84:   have hsum : ‖F - G‖ + ‖symmetricSpike p (a - b)‖ ≤ ‖F - G‖ + 2 * |a - b| := by
85:     linarith only [hsp]
86:   have hfin : ‖F - G‖ + 2 * |a - b| ≤ 3 * ‖F - G‖ := by linarith only [hab]
87:   have hn := congrArg (fun X : L2Seq => ‖X‖) he
88:   exact hn.le.trans (htri.trans (hsum.trans hfin))

91: /-- The paired lift lies on exactly one sphere for every datum in the ball. -/
92: theorem pairedSphereLift_norm {R : ℝ} (hR : 0 < R) {p : ℤ} (hp : p ≠ 0)
93:     (F : L2Seq) (hF : ‖F‖ ≤ R) (hFp : F p = 0) (hFm : F (-p) = 0) :
94:     ‖pairedSphereLift p R F‖ = 2 * R := by
95:   have hrad : 0 ≤ ((2 * R) ^ 2 - ‖F‖ ^ 2) / 2 := by nlinarith [norm_nonneg F]
96:   have hs := Real.sq_sqrt hrad
97:   have hn := norm_add_symmetricSpike_sq hp
98:     (Real.sqrt (((2 * R) ^ 2 - ‖F‖ ^ 2) / 2)) F hFp hFm
99:   change ‖pairedSphereLift p R F‖ ^ 2 = ‖F‖ ^ 2 +
100:     2 * Real.sqrt (((2 * R) ^ 2 - ‖F‖ ^ 2) / 2) ^ 2 at hn
101:   nlinarith [norm_nonneg (pairedSphereLift p R F)]

103: theorem pairedSphereLift_coeff {p k : ℤ} (hkp : k ≠ p) (hkm : k ≠ -p)
104:     (R : ℝ) (F : L2Seq) : pairedSphereLift p R F k = F k := by
105:   simp [pairedSphereLift, symmetricSpike, lp.single_apply, Ne.symm hkp, Ne.symm hkm]

108: /-- Actual homogeneous data with the paired high-frequency lift. -/
109: def sourceSphereData (s R : ℝ) (m : ℕ) (f : Fourier) (hf : InHomSobolev s f) : Fourier :=
110:   homogeneousReconstruct s (pairedSphereLift ((m : ℤ) + 1) R
111:     (homogeneousLp s (cutoff m f) (hf.cutoff m)))

113: theorem sourceSphereData_properties {s R : ℝ} (hR : 0 < R) (m : ℕ)
114:     (f : Fourier) (hf : InHomSobolev s f) (hfr : RealValued f)
115:     (hfR : homogeneousNorm s f ≤ R) :
116:     RealValued (sourceSphereData s R m f hf) ∧
117:     InHomSobolev s (sourceSphereData s R m f hf) ∧
118:     homogeneousNorm s (sourceSphereData s R m f hf) = 2 * R ∧
119:     cutoff m (sourceSphereData s R m f hf) = cutoff m f := by
120:   let p : ℤ := (m : ℤ) + 1
121:   let F := homogeneousLp s (cutoff m f) (hf.cutoff m)
122:   let Q := pairedSphereLift p R F
123:   have hp : p ≠ 0 := by dsimp [p]; omega
124:   have hpo : p ∉ Finset.Icc (-(m : ℤ)) (m : ℤ) := by simp [p]
125:   have hmo : -p ∉ Finset.Icc (-(m : ℤ)) (m : ℤ) := by simp [p]
126:   have hFp : F p = 0 := by
127:     change homogeneousWeighted s (cutoff m f) p = 0
128:     rw [homogeneousWeighted_cutoff]
129:     simp [cutoff, hpo]
130:   have hFm : F (-p) = 0 := by
131:     change homogeneousWeighted s (cutoff m f) (-p) = 0
132:     rw [homogeneousWeighted_cutoff]
133:     simp [cutoff, hmo]
134:   have hF0 : F 0 = 0 := by
135:     change (homogeneousWeight s 0 : ℂ) * cutoff m f 0 = 0
136:     rw [cutoff_meanZero hf.1 m, mul_zero]
137:   have hQ0 : Q 0 = 0 := by
138:     change F 0 + symmetricSpike p _ 0 = 0
139:     rw [hF0, symmetricSpike_zero hp, add_zero]
140:   have hQr : RealValued Q.val :=
141:     (homogeneousWeighted_real s (cutoff_real hfr m)).add (symmetricSpike_real p _)
142:   have hFN : ‖F‖ ≤ R := by
143:     rw [homogeneousLp_norm]
144:     exact (homogeneousNorm_cutoff_le hf m).trans hfR
145:   refine ⟨homogeneousReconstruct_real s Q hQr,
146:     homogeneousReconstruct_inHomSobolev s Q hQ0, ?_, ?_⟩
147:   · change homogeneousNorm s (homogeneousReconstruct s Q) = _
148:     rw [homogeneousReconstruct_norm s Q hQ0]
149:     exact pairedSphereLift_norm hR hp F hFN hFp hFm
150:   · funext k
151:     by_cases hki : k ∈ Finset.Icc (-(m : ℤ)) (m : ℤ)
152:     · simp only [cutoff, if_pos hki]
153:       by_cases hk : k = 0
154:       · subst k
155:         have hf0 : f 0 = 0 := hf.1
156:         simp [sourceSphereData, homogeneousReconstruct, hf0]
157:       · have hkp : k ≠ p := by intro he; subst k; exact hpo hki
158:         have hkm : k ≠ -p := by intro he; subst k; exact hmo hki
159:         have hQk : Q k = (homogeneousWeight s k : ℂ) * f k := by
160:           rw [pairedSphereLift_coeff hkp hkm]
161:           change (homogeneousWeight s k : ℂ) * cutoff m f k = _
162:           simp [cutoff, hki]
163:         change (if k = 0 then 0 else (homogeneousWeight s k : ℂ)⁻¹ * Q k) = f k
164:         rw [if_neg hk, hQk, ← mul_assoc,
165:           inv_mul_cancel₀ (Complex.ofReal_ne_zero.mpr (homogeneousWeight_pos_of_ne_zero s hk).ne'), one_mul]
166:     · simp [cutoff, hki]

174: theorem sourceSphereData_difference {s R : ℝ} (hR : 0 < R) (m : ℕ)
175:     (f g : Fourier) (hf : InHomSobolev s f) (hg : InHomSobolev s g)
176:     (hfR : homogeneousNorm s f ≤ R) (hgR : homogeneousNorm s g ≤ R) :
177:     homogeneousNorm s (sourceSphereData s R m f hf - sourceSphereData s R m g hg) ≤
178:       3 * homogeneousNorm s (f - g) := by
179:   let p : ℤ := (m : ℤ) + 1
180:   let F := homogeneousLp s (cutoff m f) (hf.cutoff m)
181:   let G := homogeneousLp s (cutoff m g) (hg.cutoff m)
182:   let QF := pairedSphereLift p R F
183:   let QG := pairedSphereLift p R G
184:   have hp : p ≠ 0 := by dsimp [p]; omega
185:   have hQ0 (H : L2Seq) (hH : H 0 = 0) : pairedSphereLift p R H 0 = 0 := by
186:     change H 0 + symmetricSpike p _ 0 = 0
187:     rw [hH, symmetricSpike_zero hp, add_zero]
188:   have hF0 : F 0 = 0 := by
189:     change (homogeneousWeight s 0 : ℂ) * cutoff m f 0 = 0
190:     rw [cutoff_meanZero hf.1, mul_zero]
191:   have hG0 : G 0 = 0 := by
192:     change (homogeneousWeight s 0 : ℂ) * cutoff m g 0 = 0
193:     rw [cutoff_meanZero hg.1, mul_zero]
194:   have hD0 : (QF - QG) 0 = 0 := by
195:     change QF 0 - QG 0 = 0
196:     rw [hQ0 F hF0, hQ0 G hG0, sub_self]
197:   have hFN : ‖F‖ ≤ R := by
198:     rw [homogeneousLp_norm]
199:     exact (homogeneousNorm_cutoff_le hf m).trans hfR
200:   have hGN : ‖G‖ ≤ R := by
201:     rw [homogeneousLp_norm]
202:     exact (homogeneousNorm_cutoff_le hg m).trans hgR
203:   have hDiff : F - G = homogeneousLp s (cutoff m (f - g)) ((hf.sub hg).cutoff m) := by
204:     apply Subtype.ext
205:     change homogeneousWeighted s (cutoff m f) - homogeneousWeighted s (cutoff m g) = _
206:     rw [← homogeneousWeighted_sub, ← cutoff_sub]
207:     rfl
208:   have hNormDiff : ‖F - G‖ ≤ homogeneousNorm s (f - g) := by
209:     rw [hDiff, homogeneousLp_norm]
210:     exact homogeneousNorm_cutoff_le (hf.sub hg) m
211:   change homogeneousNorm s (homogeneousReconstruct s QF - homogeneousReconstruct s QG) ≤ _
212:   rw [← homogeneousReconstruct_sub, homogeneousReconstruct_norm s (QF - QG) hD0]
213:   have hb := pairedSphereLift_lipschitz hR p F G hFN hGN
214:   exact hb.trans (by linarith only [hNormDiff])

217: theorem negative_weight_tendsto_zero {s : ℝ} (hs : 0 < s) :
218:     Filter.Tendsto (fun n : ℕ => weight (-s) (n : ℤ)) Filter.atTop (nhds 0) := by
219:   have hb : Filter.Tendsto (fun n : ℕ => 1 + (n : ℝ) ^ 2) Filter.atTop Filter.atTop := by
220:     apply Filter.tendsto_atTop_mono (f := fun n : ℕ => (n : ℝ))
221:     · intro n
222:       have hn : 0 ≤ (n : ℝ) := Nat.cast_nonneg n
223:       nlinarith [sq_nonneg ((n : ℝ) - 1 / 2)]
224:     · exact tendsto_natCast_atTop_atTop
225:   have hw := (tendsto_rpow_neg_atTop (show 0 < s / 2 by linarith)).comp hb
226:   simpa [weight, neg_div, Function.comp_def] using hw

228: /-- Padding at frequencies escaping to infinity disappears in L2 for every
229: strictly positive Sobolev exponent, even though its H^s size stays fixed. -/
230: theorem sourceSphereData_L2_converges {s R : ℝ} (hs : 0 < s) (hR : 0 < R)
231:     (f : Fourier) (hf : InHomSobolev s f) (hfr : RealValued f)
232:     (hfR : homogeneousNorm s f ≤ R) :
233:     Filter.Tendsto (fun n : ℕ => homogeneousNorm 0 (sourceSphereData s R n f hf - f))
234:       Filter.atTop (nhds 0) := by
235:   let D : ℕ → Fourier := fun n => sourceSphereData s R n f hf - f
236:   let C : ℝ := (2 : ℝ) ^ s * (3 * R)
237:   have hb (n : ℕ) : hNorm 0 (D n) ≤ weight (-s) (n : ℤ) * C := by
238:     obtain ⟨_, hpad, hnorm, hlow⟩ := sourceSphereData_properties hR n f hf hfr hfR
239:     have hD : InHomSobolev s (D n) := hpad.sub hf
240:     have hcut : cutoff n (D n) = 0 := by
241:       change cutoff n (sourceSphereData s R n f hf - f) = 0
242:       rw [cutoff_sub, hlow, sub_self]
243:     have hhigh : highCutoff n (D n) = D n := by simp [highCutoff, hcut]
244:     have hHs : hNorm s (D n) ≤ C := by
245:       have hd := homogeneousNorm_sub_le hpad hf
246:       change homogeneousNorm s (D n) ≤ _ at hd
247:       have hbd : homogeneousNorm s (D n) ≤ 3 * R := by rw [hnorm] at hd; linarith
248:       exact (hNorm_le_homogeneousNorm hs.le hD).trans
249:         (mul_le_mul_of_nonneg_left hbd (Real.rpow_nonneg (by norm_num) _))
250:     have hh := (high_cutoff_bound hs.le (hD.to_inhomogeneous hs.le) n).2
251:     rw [hhigh] at hh
252:     simp only [zero_sub] at hh
253:     exact hh.trans (mul_le_mul_of_nonneg_left hHs (weight_pos _ _).le)
254:   have hw := (negative_weight_tendsto_zero hs).mul_const C
255:   have he := squeeze_zero (fun n => hNorm_nonneg 0 (D n)) hb (by simpa using hw)
256:   simpa only [homogeneousNorm_zero] using he
