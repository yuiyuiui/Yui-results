7: /-- Only finitely many modes move. Source H0 continuity therefore preserves
8: every higher initial Sobolev regularity; no new source assumption is used. -/
9: theorem galerkin_higher_continuity {s T : ℝ} (hs : 0 ≤ s) {m : ℕ}
10:     {f : Fourier} (hf : InHomSobolev s f) {v : ℝ → Fourier}
11:     (hc : HomSobolevContinuousOn 0 T v) (hv : SourceGalerkinODE m T f v) :
12:     HomSobolevContinuousOn s T v := by
13:   have htail (t : ℝ) (ht : t ∈ Set.Icc 0 T) := hv.tail_stationary (le_refl (0 : ℝ)) hc ht
14:   have hlow (t : ℝ) (ht : t ∈ Set.Icc 0 T) : InHomSobolev s (cutoff m (v t)) :=
15:     (cutoff_inSobolev_all s m (v t)).to_homogeneous hs (cutoff_meanZero (hc.1 t ht).1 m)
16:   have hmem (t : ℝ) (ht : t ∈ Set.Icc 0 T) : InHomSobolev s (v t) := by
17:     have he : v t = cutoff m (v t) + highCutoff m f := by
18:       rw [← htail t ht]
19:       unfold highCutoff
20:       abel
21:     rw [he]
22:     exact (hlow t ht).add (hf.sub (hf.cutoff m))
23:   refine ⟨hmem, ?_⟩
24:   intro t ht ε hε
25:   obtain ⟨δ, hδ, hd⟩ := hc.2 t ht (ε / weight s (m : ℤ)) (div_pos hε (weight_pos _ _))
26:   refine ⟨δ, hδ, ?_⟩
27:   intro r hr hrt
28:   have he : v r - v t = cutoff m (v r - v t) := by
29:     rw [cutoff_sub]
30:     calc
31:       _ = (cutoff m (v r) + highCutoff m (v r)) -
32:           (cutoff m (v t) + highCutoff m (v t)) := by unfold highCutoff; abel
33:       _ = _ := by rw [htail r hr, htail t ht]; abel
34:   have hD : InHomSobolev 0 (v r - v t) := (hc.1 r hr).sub (hc.1 t ht)
35:   have hg := cutoff_gain_bound hs (hD.to_inhomogeneous (le_refl 0)) m
36:   simp only [zero_add] at hg
37:   have hn := homogeneousNorm_le_hNorm hs hg.1 (cutoff_meanZero hD.1 m)
38:   have hb : homogeneousNorm s (v r - v t) ≤ weight s (m : ℤ) * homogeneousNorm 0 (v r - v t) := by
39:     calc
40:       _ = homogeneousNorm s (cutoff m (v r - v t)) := congrArg (homogeneousNorm s) he
41:       _ ≤ hNorm s (cutoff m (v r - v t)) := hn
42:       _ ≤ _ := by simpa only [homogeneousNorm_zero] using hg.2
43:   have hh := mul_lt_mul_of_pos_left (hd r hr hrt) (weight_pos s (m : ℤ))
44:   have heps : weight s (m : ℤ) * (ε / weight s (m : ℤ)) = ε := by
45:     field_simp [(weight_pos s (m : ℤ)).ne']
46:   rw [heps] at hh
47:   exact hb.trans_lt hh

49: theorem source_galerkin_family (h41 : BITProposition41) {s T : ℝ}
50:     (hs : 0 ≤ s) (hT : 0 < T) (f : Fourier) (hfr : RealValued f) (hf : InHomSobolev s f) :
51:     ∃ G : ℕ → ℝ → Fourier, ∀ m, HomSobolevContinuousOn s T (G m) ∧
52:       (∀ t ∈ Set.Icc 0 T, RealValued (G m t)) ∧ SourceGalerkinODE (m + 1) T f (G m) := by
53:   choose G hG hEnergy using (fun m : ℕ => h41 f hfr (hf.of_le hs) (m + 1) (by omega))
54:   refine ⟨G, ?_⟩
55:   intro m
56:   obtain ⟨hc, hr, hODE⟩ := hG m T hT
57:   exact ⟨galerkin_higher_continuity hs hf hc hODE, hr, hODE⟩

60: /-- Two directions of the source Galerkin solution-class correspondence. -/
61: theorem galerkin_continuity_iff {s T : ℝ} (hs : 0 ≤ s) {m : ℕ}
62:     {f : Fourier} (hf : InHomSobolev s f) {v : ℝ → Fourier}
63:     (hv : SourceGalerkinODE m T f v) :
64:     HomSobolevContinuousOn s T v ↔ HomSobolevContinuousOn 0 T v :=
65:   ⟨fun hc => hc.of_le hs, fun hc => galerkin_higher_continuity hs hf hc hv⟩

68: /-- Solving the source Fourier sum constraint does not change its index domain. -/
69: def tripleSumEquiv (k : ℤ) : (ℤ × ℤ) ≃
70:     {q : ℤ × ℤ × ℤ // q.1 + q.2.1 + q.2.2 = k} where
71:   toFun q := ⟨(q.1, q.2, k - q.1 - q.2), by dsimp; omega⟩
72:   invFun q := (q.val.1, q.val.2.1)
73:   left_inv _ := rfl
74:   right_inv := by
75:     rintro ⟨⟨a, b, c⟩, h⟩
76:     apply Subtype.ext
77:     dsimp only
78:     apply Prod.ext
79:     · rfl
80:     · apply Prod.ext
81:       · rfl
82:       · dsimp at h ⊢
83:         omega

85: def quadrupleSumEquiv (k : ℤ) : (ℤ × ℤ × ℤ) ≃

87:   toFun q := ⟨(q.1, q.2.1, q.2.2, k - q.1 - q.2.1 - q.2.2), by dsimp; omega⟩
88:   invFun q := (q.val.1, q.val.2.1, q.val.2.2.1)
89:   left_inv _ := rfl
90:   right_inv := by
91:     rintro ⟨⟨a, b, c, d⟩, h⟩

157: /-- The differentiated old mode is c+d. Its nonzero condition survives the
158: cancellation producing (3.28), in addition to all nonresonant denominators. -/
159: theorem quartic_nonres_domain (a b c d : ℤ) :
160:     (a ≠ 0 ∧ b ≠ 0 ∧ c ≠ 0 ∧ d ≠ 0 ∧ c + d ≠ 0 ∧
161:       c + d + a ≠ 0 ∧ a + b ≠ 0 ∧ b + c + d ≠ 0) ↔
162:     (a ≠ 0 ∧ b ≠ 0 ∧ c ≠ 0 ∧ d ≠ 0 ∧ c + d ≠ 0 ∧
163:       (a + b) * (a + c + d) * (b + c + d) ≠ 0) := by
164:   rw [show c + d + a = a + c + d by ring]
165:   simp only [mul_ne_zero_iff]
166:   tauto
