57: theorem weakTimeDerivative_iff_bochner {α a b : ℝ} (hα : 0 ≤ α) (hab : a ≤ b)
58:     {f : ℝ → Sobolev α} {df : ℝ → Sobolev (-pStar)}
59:     (hf : MemLp f ∞ (volume.restrict (Set.Icc a b)))
60:     (hd : MemLp df ∞ (volume.restrict (Set.Icc a b))) :
61:     WeakTimeDerivative a b (fun t => (f t).val) (fun t => (df t).val) ↔
62:       BochnerWeakTimeDerivative hα a b f df := by
63:   let F : ℝ → Sobolev (-pStar) := fun t => inclusionMap (-pStar) α (by dsimp [pStar]; linarith) (f t)
64:   have hF : MemLp F ∞ (volume.restrict (Set.Icc a b)) :=
65:     (inclusionMap (-pStar) α (by dsimp [pStar]; linarith)).comp_memLp' hf
66:   constructor
67:   · intro h φ hφ hcomp hsupp
68:     have hD : Continuous (fun t : ℝ => ((deriv φ t : ℝ) : ℂ)) :=
69:       Complex.continuous_ofReal.comp (hφ.continuous_deriv (by simp))
70:     have hV : Continuous (fun t : ℝ => ((φ t : ℝ) : ℂ)) := Complex.continuous_ofReal.comp hφ.continuous
71:     apply Sobolev.ext
72:     funext k
73:     change (∫ t : ℝ in a..b, ((deriv φ t : ℝ) : ℂ) • F t).val k =
74:       -(∫ t : ℝ in a..b, ((φ t : ℝ) : ℂ) • df t).val k
75:     rw [smul_interval_coefficient hab hF hD k, smul_interval_coefficient hab hd hV k]
76:     change (∫ t : ℝ in a..b, ((deriv φ t : ℝ) : ℂ) * (f t).val k) =
77:       -(∫ t : ℝ in a..b, ((φ t : ℝ) : ℂ) * (df t).val k)
78:     exact h k φ hφ hcomp hsupp
79:   · intro h k φ hφ hcomp hsupp
80:     have hD : Continuous (fun t : ℝ => ((deriv φ t : ℝ) : ℂ)) :=
81:       Complex.continuous_ofReal.comp (hφ.continuous_deriv (by simp))
82:     have hV : Continuous (fun t : ℝ => ((φ t : ℝ) : ℂ)) := Complex.continuous_ofReal.comp hφ.continuous
83:     have hh := congrArg (fun z : Sobolev (-pStar) => z.val k) (h φ hφ hcomp hsupp)
84:     change (∫ t : ℝ in a..b, ((deriv φ t : ℝ) : ℂ) • F t).val k =
85:       -(∫ t : ℝ in a..b, ((φ t : ℝ) : ℂ) • df t).val k at hh
86:     rw [smul_interval_coefficient hab hF hD k, smul_interval_coefficient hab hd hV k] at hh
87:     exact hh
