8: /-- Classical smooth solutions of (1), with the initial trace specified
9: separately. The PDE is differentiated in time; no Duhamel equation is a field. -/
10: structure PaperSmoothSolution (T : ℝ) (f : Fourier) (u : ℝ → Fourier) : Prop where
11:   initial : u 0 = f
12:   spatial : ∀ s : ℝ, SobolevContinuousOn s T u
13:   temporal : ∀ k : ℤ, ∀ m : ℕ, ContDiffOn ℝ m (fun t => u t k) (Set.Icc 0 T)
14:   equation : ∀ t ∈ Set.Ioo 0 T, ∀ k : ℤ,
15:     HasDerivAt (fun r => u r k) (physicalRHS (u t) k) t
16:   reality : ∀ t ∈ Set.Icc 0 T, RealValued (u t) ∧ MeanZero (u t)

18: /-- Variation of constants for the classical equation, on the entire closed
19: interval, including the independently supplied initial trace. -/
20: theorem classical_to_duhamel {T : ℝ} {f : Fourier} {u : ℝ → Fourier}
21:     (hT : 0 ≤ T) (hc : SobolevContinuousOn 0 T u) (hi : u 0 = f)
22:     (hd : ∀ t ∈ Set.Ioo 0 T, ∀ k : ℤ,
23:       HasDerivAt (fun r => u r k) (physicalRHS (u t) k) t) : Duhamel T f u := by
24:   let v := fun t => airy (-t) (u t)
25:   have hvC : SobolevContinuousOn 0 T v := hc.airy_path (fun t : ℝ => -t) continuous_neg
26:   have hv0 : v 0 = f := by simp [v, hi]
27:   have hvE : TwistedEquation T f v := by
28:     refine ⟨hv0, ?_⟩
29:     intro t ht k
30:     have hcont : ContinuousOn (fun r => v r k) (Set.Icc 0 t) :=
31:       (hvC.coeff_continuousOn k).mono (Set.Icc_subset_Icc le_rfl ht.2)
32:     let U := sobolevExtension hT u hc
33:     have hUC : Continuous U := continuous_sobolevExtension hT _
34:     have hUv (r : ℝ) (hr : r ∈ Set.Icc 0 T) : (U r).val = u r :=
35:       sobolevExtension_val_on hT _ hr
36:     have hprod : ContinuousOn (fun r => product (u r) (u r) k) (Set.Icc 0 t) := by
37:       apply ((continuous_productCoefficientMap k hUC hUC).continuousOn).congr
38:       intro r hr
39:       change product (u r) (u r) k = product (U r).val (U r).val k
40:       rw [hUv r ⟨hr.1, hr.2.trans ht.2⟩]
41:     let E := fun r : ℝ => Complex.exp (Complex.I * ((-r : ℝ) : ℂ) * (k : ℂ) ^ 3)
42:     let D := fun r : ℝ => (1 / 2 : ℂ) * E r * (Complex.I * (k : ℂ)) * product (u r) (u r) k
43:     have hEC : Continuous E := by dsimp [E]; fun_prop
44:     have hDC : ContinuousOn D (Set.Icc 0 t) :=
45:       (((continuous_const.mul hEC).mul continuous_const).continuousOn).mul hprod
46:     have hder (r : ℝ) (hr : r ∈ Set.Ioo 0 t) : HasDerivAt (fun r => v r k) (D r) r := by
47:       have hE : HasDerivAt E (-(Complex.I * (k : ℂ) ^ 3) * E r) r := by
48:         have h := (((((hasDerivAt_id r).neg).ofReal_comp).const_mul Complex.I).mul_const ((k : ℂ) ^ 3)).cexp
49:         simpa only [E, Pi.neg_apply, id_eq, Complex.ofReal_neg, Complex.ofReal_one, mul_neg, mul_one, neg_mul,
50:           mul_comm] using h
51:       have h := hE.mul (hd r ⟨hr.1, hr.2.trans_le ht.2⟩ k)
52:       change HasDerivAt (fun r => E r * u r k) (D r) r
53:       apply h.congr_deriv
54:       rw [physicalRHS_coefficient]
55:       dsimp [D]
56:       ring
57:     have hFTC := intervalIntegral.integral_eq_sub_of_hasDerivAt_of_le ht.1 hcont hder
58:       (ContinuousOn.intervalIntegrable_of_Icc ht.1 hDC)
59:     have hD : (∫ r : ℝ in (0 : ℝ)..t, D r) = (1 / 2 : ℂ) * ∫ r : ℝ in (0 : ℝ)..t, B r (v r) (v r) k := by
60:       rw [← intervalIntegral.integral_const_mul]
61:       apply intervalIntegral.integral_congr
62:       intro r _
63:       dsimp [D, E, B, v, airy, derivative]
64:       rw [airy_inverse_neg]
65:       ring
66:     rw [hD, hv0] at hFTC
67:     linear_combination -hFTC
68:   have h := untwist_Duhamel hvE
69:   simpa only [v, airy_inverse_neg] using h

84: /-- Smooth approximation in the paper's initial H^γ and uniform solution L²
85: topologies, using classical PDE solutions rather than integral equations. -/
86: def PaperSmoothApproximation (γ T : ℝ) (f : Fourier) (u : ℝ → Fourier) : Prop :=
87:   ∃ data : ℕ → Fourier, ∃ solutions : ℕ → ℝ → Fourier,
88:     (∀ m, PaperSmoothSolution T (data m) (solutions m)) ∧
89:     Filter.Tendsto (fun m => hNorm γ (data m - f)) Filter.atTop (nhds 0) ∧
90:     ∀ ε : ℝ, 0 < ε → ∃ m₀ : ℕ, ∀ m ≥ m₀,
91:       ∀ t ∈ Set.Icc 0 T, hNorm 0 (solutions m t - u t) < ε

93: theorem paperApproximation_iff_historical {γ T : ℝ} (hT : 0 ≤ T) (f : Fourier) (u : ℝ → Fourier) :

100: /-- The paper's spatial distribution pairing, with its normalized measure:
101: Fourier coefficients act bilinearly on the reflected test coefficients. -/
102: def paperPair (a ψ : Fourier) : ℂ := ∑' k : ℤ, a k * ψ (-k)
103: 
104: /-- Distributional equation (1), tested against every smooth spatial test and
105: every compactly supported smooth time test. All solutions are evaluated in
106: their continuous representatives by the surrounding solution definition. -/
107: def PaperDistributionalKdV (T : ℝ) (u : ℝ → Fourier) : Prop :=
108:   ∀ φ : ℝ → ℝ, ContDiff ℝ (⊤ : WithTop ℕ∞) φ → HasCompactSupport φ →
109:     Function.support φ ⊆ Set.Ioo 0 T →
110:     ∀ ψ : Fourier, (∀ s : ℝ, InSobolev s ψ) →
111:       (∫ t : ℝ in (0 : ℝ)..T, ((deriv φ t : ℝ) : ℂ) * paperPair (u t) ψ) =
112:         -(∫ t : ℝ in (0 : ℝ)..T, (φ t : ℂ) * paperPair (physicalRHS (u t)) ψ)

114: /-- The independent continuous distributional/initial-trace/approximation
115: convention in Section 2. Neither this definition nor its classical approximants
116: assume Duhamel, stability, or a convergence rate. -/
117: structure PaperSolution (γ T : ℝ) (f : Fourier) (u : ℝ → Fourier) : Prop where
118:   continuous : SobolevContinuousOn γ T u
119:   initial : u 0 = f
120:   reality : ∀ t ∈ Set.Icc 0 T, RealValued (u t) ∧ MeanZero (u t)
121:   equation : PaperDistributionalKdV T u
122:   approximation : PaperSmoothApproximation γ T f u

124: theorem kdv_to_paper_distribution {γ T : ℝ} {f : Fourier} {u : ℝ → Fourier}
125:     (hT : 0 ≤ T) (hγ : 0 ≤ γ) (hu : KdVSolution γ T f u) : PaperDistributionalKdV T u := by
126:   obtain ⟨U, F, hUC, hFC, hval, hweak⟩ := kdv_solution_distributional_spatial_tests hT hγ hu
127:   intro φ hφ hsupp hsupport ψ hψ
128:   have hh := hweak φ hφ hsupp hsupport (⟨ψ, hψ 3⟩ : Sobolev 3)
129:   have he1 : (∫ t : ℝ in (0 : ℝ)..T, ((deriv φ t : ℝ) : ℂ) *
130:       distributionTest 3 ⟨ψ, hψ 3⟩ (inclusionMap (-3) 0 (by norm_num) (U t))) =
131:       ∫ t : ℝ in (0 : ℝ)..T, ((deriv φ t : ℝ) : ℂ) * paperPair (u t) ψ := by
132:     apply intervalIntegral.integral_congr
133:     intro t ht
134:     rw [Set.uIcc_of_le hT] at ht
135:     dsimp only
136:     rw [distributionTest_apply]
137:     change _ * (∑' k : ℤ, (U t).val k * ψ (-k)) = _
138:     rw [(hval t ht).1]
139:     rfl
140:   have he2 : (∫ t : ℝ in (0 : ℝ)..T, (φ t : ℂ) * distributionTest 3 ⟨ψ, hψ 3⟩ (F t)) =
141:       ∫ t : ℝ in (0 : ℝ)..T, (φ t : ℂ) * paperPair (physicalRHS (u t)) ψ := by
142:     apply intervalIntegral.integral_congr
143:     intro t ht
144:     rw [Set.uIcc_of_le hT] at ht
145:     dsimp only
146:     rw [distributionTest_apply, (hval t ht).2]
147:     rfl
148:   rwa [he1, he2] at hh

150: /-- Integral equations depend only on the closed-interval representative. -/
151: theorem duhamel_congr_on {T : ℝ} (hT : 0 ≤ T) {f : Fourier} {u v : ℝ → Fourier}

163: /-- Smooth approximation determines the same continuous representative as the
164: exact flow. This comparison uses exact-flow theorems only, and assumes no PDE
165: or integral equation for the limit path. -/
166: theorem approximation_determines_flow
167:     (h41 : BITProposition41) (h43 : BITTheorems42And43)
168:     (h51 : BITTheorem51) (h63 : BITTheorem63) (h65 : BITTheorem65)
169:     {γ T : ℝ} (hγ : 0 < γ) (hT : 0 < T) {f : Fourier}
170:     (hf : InSobolev γ f) (hfr : RealValued f) (hfm : MeanZero f)
171:     {u : ℝ → Fourier} (hc : SobolevContinuousOn γ T u)
172:     (ha : ObtainedBySmoothApproximation γ T f u) : KdVSolution γ T f u := by
173:   obtain ⟨v, hv⟩ := positive_kdv_flow_exists
174:     (galerkin_to_historical (bit42_from_joint h43)) (limit_to_historical h41 h43)
175:     (high_to_historical h41 h43 h51 h65) (low_to_historical h41 h43 h63 h65)
176:     hγ hT hf hfr hfm
177:   obtain ⟨data, solutions, hsol, hdata, happ⟩ := ha
178:   have hzero : (0 : ℝ) ∈ Set.Icc 0 T := ⟨le_rfl, hT.le⟩
179:   have hdγ (m : ℕ) : InSobolev γ (data m) := by
180:     have hh := (hsol m).2.1 γ |>.1 0 hzero
181:     rwa [(hsol m).1.1] at hh
182:   have hd0 (m : ℕ) := (hdγ m).of_le hγ.le
183:   have hdr (m : ℕ) : RealValued (data m) ∧ MeanZero (data m) := by
184:     simpa only [(hsol m).1.1] using (hsol m).2.2.2 0 hzero
185:   have hdif (m : ℕ) := (hdγ m).sub hf
186:   have hdata0 : Filter.Tendsto (fun m => hNorm 0 (data m - f)) Filter.atTop (nhds 0) :=
187:     squeeze_zero (fun m => hNorm_nonneg _ _) (fun m => hNorm_mono (hdif m) hγ.le) hdata
188:   obtain ⟨C, hC, hdep⟩ := kdv_flow_l2_lipschitz h65
189:     (show 0 ≤ 2 * hNorm 0 f + 1 by linarith [hNorm_nonneg 0 f]) hT
190:   have hbound : ∀ᶠ m : ℕ in Filter.atTop, hNorm 0 (data m) + hNorm 0 f ≤ 2 * hNorm 0 f + 1 := by
191:     filter_upwards [hdata0.eventually (gt_mem_nhds (by norm_num : (0 : ℝ) < 1))] with m hm
192:     have hh : hNorm 0 (data m) ≤ hNorm 0 (data m - f) + hNorm 0 f := by
193:       simpa only [sub_add_cancel] using hNorm_add_le ((hd0 m).sub (hf.of_le hγ.le)) (hf.of_le hγ.le)
194:     linarith
195:   have heq : ∀ t ∈ Set.Icc 0 T, u t = v t := by
196:     intro t ht
197:     have hun := (hc.1 t ht).of_le hγ.le
198:     have hvn := (hv.1.1 t ht).of_le hγ.le
199:     have hlim : Filter.Tendsto (fun m => C * hNorm 0 (data m - f)) Filter.atTop (nhds 0) := by
200:       simpa only [mul_zero] using (tendsto_const_nhds (x := C)).mul hdata0
201:     have hz : hNorm 0 (u t - v t) ≤ 0 := by
202:       apply le_of_forall_pos_le_add
203:       intro ε hε
204:       obtain ⟨m₀, hm₀⟩ := happ (ε / 2) (by positivity)
205:       have hevent := hlim.eventually (gt_mem_nhds (show (0 : ℝ) < ε / 2 by positivity))
206:       obtain ⟨m, hmB, hmE, hmm⟩ := (hbound.and (hevent.and (Filter.eventually_ge_atTop m₀))).exists
207:       have hsm : KdVSolution γ T (data m) (solutions m) :=
208:         ⟨(hsol m).2.1 γ, (hsol m).1, (hsol m).2.2.2⟩
209:       have hcomp := hdep γ γ hγ.le hγ.le (data m) f (hdr m).1 (hdr m).2 (hdγ m)
210:         hfr hfm hf hmB (solutions m) v hsm hv t ht
211:       have hsn := ((hsol m).2.1 0).1 t ht
212:       have hsplit : u t - v t = (u t - solutions m t) + (solutions m t - v t) := by abel
213:       have hh := hNorm_add_le (hun.sub hsn) (hsn.sub hvn)
214:       rw [← hsplit, hNorm_sub_symm 0 (u t) (solutions m t)] at hh
215:       have happm := hm₀ m hmm t ht
216:       linarith
217:     funext k
218:     have hk := weighted_coeff_bound (hun.sub hvn) k
219:     rw [weighted_zero] at hk
220:     exact sub_eq_zero.mp (norm_eq_zero.mp (le_antisymm (hk.trans hz) (norm_nonneg _)))
221:   exact ⟨hc, duhamel_congr_on hT.le hv.2.1 heq, fun t ht => heq t ht ▸ hv.2.2 t ht⟩

240: /-- Continuous representatives which agree almost everywhere agree at both
241: endpoints as well as in the interior. No extra trace convention is imposed. -/
242: theorem continuous_representatives_eq {γ T : ℝ} (hT : 0 < T)
243:     {u v : ℝ → Fourier} (hu : SobolevContinuousOn γ T u) (hv : SobolevContinuousOn γ T v)
244:     (he : u =ᵐ[volume.restrict (Set.Icc 0 T)] v) :
245:     ∀ t ∈ Set.Icc 0 T, u t = v t := by
246:   intro t ht
247:   funext k

249:     (hu.coeff_continuousOn k) (hv.coeff_continuousOn k) ht

253: /-- The target statement with the paper's independent solution convention.
254: Every other definition and quantifier is literally the historical target. -/
255: def PaperTheoremOne : Prop :=
256:   ∀ T : ℝ, 0 < T → ∀ γ : ℝ, 0 < γ → γ ≤ 1 → ∀ R : ℝ, 0 < R →
257:     ∃ C : ℝ, 0 < C ∧ ∃ τ₀ : ℝ, 0 < τ₀ ∧
258:       ∀ f : Fourier, InSobolev γ f → RealValued f → MeanZero f → hNorm γ f ≤ R →
259:         ∀ u : ℝ → Fourier, PaperSolution γ T f u →
260:           ∀ L : ℕ, 0 < L → T / (L : ℝ) ≤ τ₀ →
261:             maximumError T L f u ≤ C * (T / (L : ℝ)) ^ γ

263: /-- Substantive two-way target equivalence; it preserves C and τ₀ exactly,
264: the uniform data ball, the unfiltered step, and the complete time grid. -/
265: theorem theoremOne_iff_paper
266:     (h41 : BITProposition41) (h43 : BITTheorems42And43)
267:     (h51 : BITTheorem51) (h63 : BITTheorem63) (h65 : BITTheorem65) :
268:     TheoremOne ↔ PaperTheoremOne := by
269:   constructor <;> intro h T hT γ hγ hγ1 R hR
270:   · obtain ⟨C, hC, τ₀, hτ₀, hb⟩ := h T hT γ hγ hγ1 R hR
271:     refine ⟨C, hC, τ₀, hτ₀, ?_⟩
272:     intro f hf hfr hfm hfR u hu
273:     exact hb f hf hfr hfm hfR u
274:       ((paperSolution_iff_exactSolution h41 h43 h51 h63 h65 hγ hT hf hfr hfm u).mp hu)
275:   · obtain ⟨C, hC, τ₀, hτ₀, hb⟩ := h T hT γ hγ hγ1 R hR
276:     refine ⟨C, hC, τ₀, hτ₀, ?_⟩
