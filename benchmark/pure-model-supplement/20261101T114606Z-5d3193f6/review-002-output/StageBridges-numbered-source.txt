6: theorem J_shift (a r : ℝ) (f g : Fourier) :
7:     airy (-a) (J r (airy a f) (airy a g)) = J (a + r) f g := by
8:   simp only [J, inverseDerivative_airy, ← airy_add_time]
9:   rw [show -a + -r = -(a + r) by ring, show r + a = a + r by ring]

24: /-- Preserve the source's complete-grid quantification. An arbitrary shorter
25: step is embedded in a grid with the *same* T; its grid step is between tau and
26: 2*tau. Airy conjugation moves its start to grid point zero. -/
27: theorem stage_grid_to_historical (h : LiWuLemma72Grid) : LiWuStageEstimate := by
28:   intro T hT γ hγ hγ1 ε hε β hb hb1

32:   refine ⟨C * (2 : ℝ) ^ q, mul_pos hC (Real.rpow_pos_of_pos (by norm_num) _), ?_⟩
33:   intro τ n t hτ hn ht f g hf hg hfm hgm
34:   have hτT : τ ≤ T := by

40:   have hL : 0 < L := Nat.floor_pos.mpr hx

42:   have hL1 : 1 ≤ (L : ℝ) := by exact_mod_cast hL
